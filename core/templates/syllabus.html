{# ── Shell Exploration Guide ── #}

💻 Django Shell Demonstration Commands (python manage.py shell)

The following 5 standard ORM queries can be run directly inside Django shell to verify data persistence and relationship traversals:

# 1. Filter signals with confidence >= 80
from core.models import StockSignal, Trade, Portfolio, User
high_conf = StockSignal.objects.filter(ml_confidence__gte=80)
print([(s.symbol, s.ml_confidence) for s in high_conf])

# 2. Inspect user portfolio balance and related holdings
demo_user = User.objects.get(username='demo')
portfolio = demo_user.portfolio
print(f"Cash: ₹{portfolio.cash}, Equity: ₹{portfolio.total_value}")
print(list(portfolio.holdings.values('symbol', 'quantity', 'avg_price')))

# 3. Check Jev compliance execution records
executed_trades = Trade.objects.filter(decision='execute', jev_approved=True)
print(f"Total approved executions: {executed_trades.count()}")
